## 🧮 三体表决算法实现

### 核心算法

```rust
/// 三体表决决策器
pub struct ThreeBodyVoter {
    // 引力常数 (可学习)
    gravity_constant: f32,

    // 智能体配置
    agents: Vec<AgentConfig>,

    // 决策历史
    history: Vec<DecisionRecord>,
}

impl ThreeBodyVoter {
    /// 三体引力决策
    pub fn vote(&self, inputs: &[AgentInput]) -> Decision {
        let states: Vec<DecisionState> = inputs.iter()
            .map(|i| self.compute_state(i))
            .collect();

        // 1. 计算引力矩阵
        let forces = self.compute_gravitational_forces(&states);

        // 2. 求解平衡点
        let equilibrium = self.find_equilibrium(&states, &forces);

        // 3. 生成最终决策
        self.generate_decision(&equilibrium, &states)
    }

    /// 计算决策状态向量
    fn compute_state(&self, input: &AgentInput) -> DecisionState {
        let magnitude = input.confidence * input.activation;
        let direction = input.preference_vector;

        DecisionState {
            position: Vector3D::from(direction),
            mass: magnitude,           // 置信度作为"质量"
            velocity: input.derivative,
        }
    }

    /// 引力计算: F = G·m₁·m₂/r²
    fn compute_gravitational_forces(&self, states: &[DecisionState]) -> Vec<Vector3D> {
        let n = states.len();
        let mut forces = vec![Vector3D::zero(); n];

        for i in 0..n {
            for j in 0..n {
                if i != j {
                    let r_vec = states[j].position - states[i].position;
                    let r = r_vec.norm() + EPSILON; // 防止除零

                    let m1 = states[i].mass;
                    let m2 = states[j].mass;

                    // F = G * m1 * m2 / r²
                    let f_magnitude = self.gravity_constant * m1 * m2 / (r * r);

                    let f_vec = r_vec.normalize() * f_magnitude;
                    forces[i] = forces[i] + f_vec;
                }
            }
        }

        forces
    }

    /// 寻找动态平衡点
    fn find_equilibrium(&self, states: &[DecisionState], forces: &[Vector3D]) -> Vector3D {
        // 质心偏移
        let centroid: Vector3D = states.iter()
            .map(|s| s.position * s.mass)
            .sum::<Vector3D>() /
            states.iter().map(|s| s.mass).sum::<f32>();

        // 引力修正
        let force_sum: Vector3D = forces.iter().sum();

        // 平衡点 = 质心 + 扰动
        centroid + force_sum * 0.1
    }
}

/// 三体决策输入
pub struct AgentInput {
    pub agent_id: String,
    pub agent_type: AgentType,
    pub preference_vector: Vec<f32>,     // 决策倾向
    pub confidence: f32,                  // 置信度
    pub activation: f32,                  // 激活强度
    pub domain: DecisionDomain,          // 决策领域
}

/// 决策领域
pub enum DecisionDomain {
    Safety,        // 安全决策
    Efficiency,    // 效率决策
    Quality,       // 质量决策
    Comfort,       // 舒适决策
}
```

### 权重自适应机制

```rust
/// 自适应三体权重
pub struct AdaptiveThreeBody {
    base_voter: ThreeBodyVoter,
    performance_tracker: PerformanceTracker,
}

impl AdaptiveThreeBody {
    /// 根据任务类型调整权重
    pub fn adaptive_vote(&mut self, task: &Task, inputs: &[AgentInput]) -> Decision {
        // 1. 分析任务特征
        let task_weights = self.analyze_task(task);

        // 2. 加权决策输入
        let weighted_inputs: Vec<AgentInput> = inputs.iter()
            .zip(task_weights.iter())
            .map(|(input, weight)| {
                AgentInput {
                    confidence: input.confidence * weight.confidence_weight,
                    activation: input.activation * weight.activation_weight,
                    ..input.clone()
                }
            })
            .collect();

        // 3. 三体表决
        let decision = self.base_voter.vote(&weighted_inputs);

        // 4. 更新权重
        self.update_weights(&decision, task);

        decision
    }

    /// 分析任务确定权重
    fn analyze_task(&self, task: &Task) -> Vec<AdaptiveWeight> {
        match task.task_type {
            TaskType::Navigation => vec![
                AdaptiveWeight { confidence_weight: 0.2, activation_weight: 0.8 },  // PFNN
                AdaptiveWeight { confidence_weight: 0.5, activation_weight: 0.3 },  // SNN
                AdaptiveWeight { confidence_weight: 0.3, activation_weight: 0.2 },  // LLM
            ],
            TaskType::ObjectRecognition => vec![
                AdaptiveWeight { confidence_weight: 0.3, activation_weight: 0.4 },  // PFNN
                AdaptiveWeight { confidence_weight: 0.5, activation_weight: 0.5 },  // SNN
                AdaptiveWeight { confidence_weight: 0.2, activation_weight: 0.2 },  // LLM
            ],
            TaskType::Planning => vec![
                AdaptiveWeight { confidence_weight: 0.1, activation_weight: 0.1 },  // PFNN
                AdaptiveWeight { confidence_weight: 0.2, activation_weight: 0.2 },  // SNN
                AdaptiveWeight { confidence_weight: 0.7, activation_weight: 0.8 },  // LLM
            ],
        }
    }
}
```

| **核心优先级** | 安全 > 舒适 > 效率 | 质量 > 效率 > 安全 |

| **PFNN 权重** | 中 (移动为主) | 中 (装配为主) |
| **LLM 权重** | 高 (交互为主) | 高 (分析为主) |
| **表决频率** | 高频 (实时交互) | 中频 (任务决策) |
| **学习周期** | 短 (快速适应习惯) | 长 (稳定工艺参数) |

### 三体表决优势

| 优势 | 家庭场景体现 | 工厂场景体现 |
|------|--------------|--------------|
| **快速响应** | 紧急情况秒级响应 | 设备故障实时检测 |
| **稳定可靠** | 避免单点错误 | 减少误判率 |
| **自适应** | 适应家庭习惯 | 适应工艺变化 |
| **可解释** | 决策过程透明 | 根因分析清晰 |
