### 2.2 接口设计

```rust
// 优必选机器人适配层
pub struct UbtechAdapter {
    robot_type: WalkerType,
    sdk: WalkerSDK,
    snn_layer: SNNController,
    pfnn_layer: PFNNController,
}

impl RobotAgent for UbtechAdapter {
    // 视觉数据 → SNN 处理
    fn get_snn_output(&self) -> Spikes {
        let vision_data = self.sdk.get_vision();
        self.snn_layer.encode_events(vision_data)
    }

    // 运动控制 → PFNN 生成
    fn execute_pfnn_motion(&mut self, targets: JointTargets) {
        self.pfnn_layer.generate_motion(targets);
        self.sdk.send_joint_commands(targets.positions);
    }
}
```
